     405 +      # 全部子进程带 -nostdin 且 stdin=DEVNULL：避免 SIGTTIN / 管道反压挂起
     406 +      # （见 memory/project_ffmpeg_stdin_hang.md）
     407 +      return subprocess.run(list(cmd), capture_output=True, text=True,
     408 +                            encoding="utf-8", errors="replace", timeout=timeout,
     409 +                            stdin=subprocess.DEVNULL, env=_ffmpeg_env())
     410 +  
     411 +  
     412 +  def _run_ffprobe_json(ffprobe_bin: str, args: Sequence[str],
     413 +                        timeout: int = PROBE_TIMEOUT) -> dict:
     414 +      cmd = [ffprobe_bin, "-v", "error", "-print_format", "json", *args]
     415 +      r = _run(cmd, timeout)
     416 +      if r.returncode != 0:
     417 +          raise RuntimeError((r.stderr or "").strip().splitlines()[-1]
     418 +                             if (r.stderr or "").strip() else f"ffprobe rc={r.returncode}")
     419 +      return json.loads(r.stdout or "{}")
     420 +  
     421 +  
     422 +  def _extract_rotation(stream: dict) -> int:
     423 +      for sd in stream.get("side_data_list") or []:
     424 +          if "isplay Matrix" in (sd.get("side_data_type") or ""):
     425 +              try:
     426 +                  return abs(int(float(sd.get("rotation") or 0))) % 360
     427 +              except (TypeError, ValueError):
     428 +                  pass
     429 +      tags = stream.get("tags") or {}
     430 +      for key in ("rotate", "rotation"):
     431 +          if key in tags:
     432 +              try:
     433 +                  return abs(int(float(tags[key]))) % 360
     434 +              except (TypeError, ValueError):
     435 +                  pass
     436 +      return 0
     437 +  
     438 +  
     439 +  def _parse_rate(value) -> Optional[float]:
     440 +      """解析 "30000/1001" / "25" / 25.0 → float；无法解析返回 None。"""
     441 +      if value is None:
     442 +          return None
     443 +      if isinstance(value, (int, float)):
     444 +          return float(value) if value > 0 else None
     445 +      s = str(value).strip()
     446 +      if not s or s in ("0/0", "N/A"):
     447 +          return None
     448 +      if "/" in s:
     449 +          num, _, den = s.partition("/")
     450 +          try:
     451 +              num_v, den_v = float(num), float(den)
     452 +              return num_v / den_v if den_v else None
     453 +          except ValueError:
     454 +              return None
     455 +      try:
     456 +          v = float(s)
     457 +          return v if v > 0 else None
     458 +      except ValueError:
     459 +          return None
     460 +  
     461 +  
     462 +  def _parse_bits(video: dict) -> int:
     463 +      for key in ("bits_per_raw_sample", "bits_per_coded_sample"):
     464 +          v = video.get(key)
     465 +          if v:
     466 +              try:
     467 +                  b = int(v)
     468 +                  if b > 0:
     469 +                      return b
     470 +              except (TypeError, ValueError):
     471 +                  pass
     472 +      pix_fmt = video.get("pix_fmt") or ""
     473 +      for suffix, bits in (("p10le", 10), ("p12le", 12), ("p16le", 16)):
     474 +          if pix_fmt.endswith(suffix):
     475 +              return bits
     476 +      return 8
     477 +  
     478 +  
     479 +  def _looks_hdr(video: dict, bits: int) -> bool:
     480 +      transfer = (video.get("color_transfer") or "").lower()
     481 +      primaries = (video.get("color_primaries") or "").lower()
     482 +      if transfer in ("smpte2084", "arib-std-b67", "bt2020-10"):
     483 +          return True
     484 +      if transfer.startswith("bt2020") and bits >= 10:
     485 +          return True
     486 +      return primaries == "bt2020" and bits >= 10
     487 +  
     488 +  
     489 +  def _has_hdr_side_data(video: dict) -> bool:
     490 +      for sd in video.get("side_data_list") or []:
     491 +          t = (sd.get("side_data_type") or "").lower()
     492 +          if "hdr" in t or "mastering display" in t or "content light level" in t:
     493 +              return True
     494 +      return False

     496 +  
     497 +  def _normalize_container(format_name: str) -> str:
     498 +      """'mov,mp4,m4a,3gp,3g2,mj2' → 'mp4'；取主容器名。"""
     499 +      if not format_name:
     500 +          return "—"
     501 +      parts = [p.strip().lower() for p in format_name.split(",") if p.strip()]
     502 +      if not parts:
     503 +          return "—"
     504 +      # ffprobe 把主容器放首位，但 mov 家族首选展示 mov
     505 +      return parts[0]

     507 +  
     508 +  def _fmt_bitrate(bps: Optional[float]) -> str:
     509 +      if not bps or bps <= 0:
     510 +          return "—"
     511 +      if bps >= 1_000_000:
     512 +          v = bps / 1_000_000.0
     513 +          return f"{v:.1f} Mbps" if v < 100 else f"{v:.0f} Mbps"
     514 +      return f"{bps / 1000.0:.0f} kbps"
     515 +  
     516 +  
     517 +  def _fmt_duration(seconds: Optional[float]) -> str:
     518 +      if not seconds or seconds <= 0:
     519 +          return "—"
     520 +      total = int(round(seconds))
     521 +      h, rem = divmod(total, 3600)
     522 +      m, s = divmod(rem, 60)
     523 +      if h:
     524 +          return f"{h}:{m:02d}:{s:02d}"
     525 +      return f"{m:02d}:{s:02d}"
     526 +  
     527 +  
     528 +  def _fmt_fps(fps: Optional[float]) -> str:
     529 +      if not fps or fps <= 0:
     530 +          return "—"
     531 +      if abs(fps - round(fps)) < 1e-4:
     532 +          return f"{round(fps)} fps"
     533 +      return f"{fps:.3f} fps"

     536 +  def _count_frames_gpu(ffprobe_bin: str, path: str, index: int) -> Optional[int]:
     537 +      """档 2：CUDA 硬解整片解码精确数帧。"""
     538 +      cmd = [ffprobe_bin, "-nostdin", "-v", "error",
     539 +             "-hwaccel", "cuda", "-hwaccel_device", "0",
     540 +             "-select_streams", f"v:{index}", "-count_frames",
     541 +             "-show_entries", "stream=nb_read_frames",
     542 +             "-print_format", "json", path]
     543 +      r = subprocess.run(cmd, capture_output=True, text=True, timeout=FRAME_COUNT_TIMEOUT,
     544 +                         stdin=subprocess.DEVNULL, env=_ffmpeg_env())
     545 +      if r.returncode != 0:
     546 +          err = (r.stderr or "").lower()
     547 +          if any(k in err for k in _CUDA_ERROR_KEYS):
     548 +              GpuProbe.disable_runtime("硬解数帧命中 CUDA 错误")
     549 +          raise RuntimeError("硬解数帧失败")
     550 +      data = json.loads(r.stdout or "{}")
     551 +      streams = data.get("streams") or []
     552 +      if not streams:
     553 +          return None
     554 +      n = streams[0].get("nb_read_frames")
     555 +      try:
     556 +          n = int(n)
     557 +      except (TypeError, ValueError):
     558 +          return None
     559 +      return n if n > 0 else None

     562 +  def _count_packets(ffprobe_bin: str, path: str, index: int) -> Optional[int]:
     563 +      """档 3：只解复用数包，不解码（CPU 侧的降级手段）。"""
     564 +      data = _run_ffprobe_json(ffprobe_bin, [
     565 +          "-select_streams", f"v:{index}", "-count_packets",
     566 +          "-show_entries", "stream=nb_read_packets", path,
     567 +      ], timeout=FRAME_COUNT_TIMEOUT)
     568 +      streams = data.get("streams") or []
     569 +      if not streams:
     570 +          return None
     571 +      try:
     572 +          n = int(streams[0].get("nb_read_packets"))
     573 +      except (TypeError, ValueError):
     574 +          return None
     575 +      return n if n > 0 else None

     578 +  def probe_video(path: str, ffprobe_bin: str, ffmpeg_bin: str,
     579 +                  opts: "Options") -> dict:
     580 +      """探测单个视频，返回渲染所需的属性字典（失败时带 error 字段）。"""
     581 +      key = _probe_cache_key(path)
     582 +      with _PROBE_LOCK:
     583 +          cached = _PROBE_CACHE.get(key)
     584 +      if cached is not None:
     585 +          return cached
     586 +  
     587 +      info: Dict = {"error": None}
     588 +      try:
     589 +          data = _run_ffprobe_json(ffprobe_bin, [
     590 +              "-show_format", "-show_streams", path,
     591 +          ])
     592 +      except Exception as exc:                       # noqa: BLE001
     593 +          info["error"] = f"不可探测（{_short_err(exc)}）"
     594 +          with _PROBE_LOCK:
     595 +              _PROBE_CACHE[key] = info
     596 +          return info
     597 +  
     598 +      streams = data.get("streams") or []
     599 +      video = next((s for s in streams
     600 +                    if s.get("codec_type") == "video"
     601 +                    and not (s.get("disposition") or {}).get("attached_pic")), None)
     602 +      if video is None:
     603 +          video = next((s for s in streams if s.get("codec_type") == "video"), None)
     604 +      if video is None:
     605 +          info["error"] = "不可探测（无视频流）"
     606 +          with _PROBE_LOCK:
     607 +              _PROBE_CACHE[key] = info
     608 +          return info
     609 +  
     610 +      v_index = 0
     611 +      for i, s in enumerate(streams):
     612 +          if s.get("codec_type") == "video" and \
     613 +                  not (s.get("disposition") or {}).get("attached_pic"):
     614 +              v_index = i
     615 +              break
     616 +      # v:N 的 N 是视频流的序号（ffprobe 按类型编号），不是全局流索引
     617 +      v_order = 0
     618 +      for s in streams[:v_index]:
     619 +          if s.get("codec_type") == "video" and \
     620 +                  not (s.get("disposition") or {}).get("attached_pic"):
     621 +              v_order += 1
     622 +  
     623 +      rotation = _extract_rotation(video)
     624 +      width = int(video.get("width") or 0)
     625 +      height = int(video.get("height") or 0)
     626 +      if rotation in (90, 270):
     627 +          width, height = height, width
     628 +  
     629 +      fmt = data.get("format") or {}
     630 +      duration = _parse_rate(fmt.get("duration"))
     631 +      if duration is None:
     632 +          duration = _parse_rate(video.get("duration"))
     633 +      if duration is None:
     634 +          tags = video.get("tags") or {}
     635 +          duration = _parse_rate(tags.get("DURATION") or tags.get("duration"))
     636 +  
     637 +      bitrate = _parse_rate(fmt.get("bit_rate"))
     638 +      if not bitrate:
     639 +          bitrate = _parse_rate(video.get("bit_rate"))
     640 +      if not bitrate and duration and duration > 0:
     641 +          try:
     642 +              bitrate = os.path.getsize(path) * 8 / duration
     643 +          except OSError:
     644 +              bitrate = None
     645 +  
     646 +      fps = _parse_rate(video.get("avg_frame_rate")) or _parse_rate(video.get("r_frame_rate"))
     647 +  
     648 +      nb_frames = None
     649 +      try:
     650 +          nb = int(video.get("nb_frames") or 0)
     651 +          nb_frames = nb if nb > 0 else None
     652 +      except (TypeError, ValueError):
     653 +          nb_frames = None
     654 +  
     655 +      # ── 帧数四级降级链 ──
     656 +      frames: Optional[int] = None
     657 +      frames_src = "—"
     658 +      if not opts.deep:
     659 +          if nb_frames:
     660 +              frames, frames_src = nb_frames, "包头"
     661 +  
     662 +      if frames is None and not opts.fast:
     663 +          gpu_ok = (not opts.cpu) and GpuProbe.available(ffmpeg_bin, opts.verbose)
     664 +          if gpu_ok:
     665 +              try:
     666 +                  n = _count_frames_gpu(ffprobe_bin, path, v_order)
     667 +                  if n:
     668 +                      frames, frames_src = n, "硬解"
     669 +              except Exception:                       # noqa: BLE001
     670 +                  frames = None
     671 +          if frames is None:
     672 +              try:
     673 +                  n = _count_packets(ffprobe_bin, path, v_order)
     674 +                  if n:
     675 +                      frames, frames_src = n, "包数"
     676 +              except Exception:                       # noqa: BLE001
     677 +                  frames = None
     678 +  
     679 +      if frames is None and duration and fps:
     680 +          est = int(round(duration * fps))
     681 +          if est > 0:
     682 +              frames, frames_src = est, "估算"
     683 +  
     684 +      bits = _parse_bits(video)
     685 +      audios = [s for s in streams if s.get("codec_type") == "audio"]
     686 +      subs = [s for s in streams if s.get("codec_type") == "subtitle"]
     687 +  
     688 +      info.update({
     689 +          "width": width,
     690 +          "height": height,
     691 +          "fps": fps,
     692 +          "bitrate": bitrate,
     693 +          "frames": frames,
     694 +          "frames_src": frames_src,
     695 +          "codec": video.get("codec_name") or "—",
     696 +          "container": _normalize_container(fmt.get("format_name") or ""),
     697 +          "duration": duration,
     698 +          "pix_fmt": video.get("pix_fmt") or "—",
     699 +          "bits": bits,
     700 +          "hdr": bool(_has_hdr_side_data(video) or _looks_hdr(video, bits)),
     701 +          "audio": _describe_audio(audios),
     702 +          "subs": _describe_subs(subs),
     703 +      })
     704 +  
     705 +      with _PROBE_LOCK:
     706 +          _PROBE_CACHE[key] = info
     707 +      return info
     708 +

     712 +      if not msg:
     713 +          msg = exc.__class__.__name__

     717 +  def _describe_audio(audios: List[dict]) -> str:
     718 +      if not audios:
     719 +          return "无"
     720 +      a = audios[0]
     721 +      codec = a.get("codec_name") or "?"
     722 +      ch = a.get("channels")
     723 +      try:
     724 +          ch_s = f"{int(ch)}ch" if ch else "?ch"
     725 +      except (TypeError, ValueError):
     726 +          ch_s = "?ch"
     727 +      rate = _parse_rate(a.get("sample_rate"))
     728 +      rate_s = f"{rate / 1000:.1f}kHz" if rate else "?kHz"
     729 +      s = f"{codec} {ch_s} {rate_s}"
     730 +      if len(audios) > 1:
     731 +          s += f" (+{len(audios) - 1})"
     732 +      return s

     852 +  def format_grid(names: Sequence[str], width: int) -> List[str]:
     853 +      """coreutils ls 的列布局：竖着填（column-major），列宽 = 该列最大显示宽。
     854 +  
     855 +      列间固定 2 空格（最后一列不加），列宽下限 MIN_COLUMN_WIDTH，
     856 +      总宽需 **严格小于** 终端宽；实在放不下（连 2 列都不行）退化为每行一个。
     857 +      """
     858 +      n = len(names)
     859 +      if n == 0:
     860 +          return []
     861 +      if n == 1:
     862 +          return [names[0]]
     863 +  
     864 +      widths = [disp_width(x) for x in names]
     865 +      chosen: Optional[Tuple[int, List[int]]] = None
     866 +      for cols in range(n, 1, -1):
     867 +          rows = (n + cols - 1) // cols
     868 +          colw: List[int] = []
     869 +          for c in range(cols):
     870 +              seg = widths[c * rows:(c + 1) * rows]
     871 +              colw.append(max(MIN_COLUMN_WIDTH, max(seg)) if seg else 0)
     872 +          total = sum(colw) + 2 * (cols - 1)
     873 +          if total < width:
     874 +              chosen = (cols, colw)
     875 +              break
     876 +  
     877 +      if chosen is None:
     878 +          return list(names)
     879 +  
     880 +      cols, colw = chosen
     881 +      rows = (n + cols - 1) // cols
     882 +      lines: List[str] = []
     883 +      for r in range(rows):
     884 +          parts: List[str] = []
     885 +          for c in range(cols):
     886 +              i = c * rows + r
     887 +              if i >= n:
     888 +                  continue
     889 +              if c == cols - 1:
     890 +                  parts.append(names[i])
     891 +              else:
     892 +                  parts.append(pad_display(names[i], colw[c]))
     893 +          lines.append("  ".join(parts).rstrip())
     894 +      return lines

     897 +  def human_size(nbytes: int) -> str:
     898 +      """coreutils 语义：base 1024，**向上取整**，<10 保留 1 位小数。"""
     899 +      n = int(nbytes)
     900 +      if n < 1024:
     901 +          return str(n)
     902 +      units = "KMGTPEZY"
     903 +      val = float(n)
     904 +      i = 0
     905 +      while val >= 1024 and i < len(units) - 1:
     906 +          val /= 1024.0
     907 +          i += 1
     908 +      sfx = units[i]
     909 +      if val < 10:
     910 +          return f"{math.ceil(val * 10) / 10:.1f}{sfx}"
     911 +      return f"{math.ceil(val)}{sfx}"

     966 +  def video_cells(entry: Entry, fields: Sequence[str]) -> Dict[str, str]:

     983 +          elif key == "codec":

     985 +          elif key == "container":

    1002 +  def render_block(entries: List[Entry], args: "Options", out) -> None:
    1003 +      """渲染一个目录块（默认分组版式或 -l 长格式）。"""
    1004 +      fields = active_fields(args)
    1005 +      videos = [e for e in entries if e.is_video]
    1006 +      video_rows = {id(e): video_cells(e, fields) for e in videos}
    1007 +  
    1008 +      colw: Dict[str, int] = {}
    1009 +      for key in fields:
    1010 +          colw[key] = max([disp_width(FIELD_HEADER[key])] +
    1011 +                          [disp_width(video_rows[id(e)].get(key, "")) for e in videos] or [1])
    1012 +      namew = max([disp_width(e.name) for e in videos], default=0)
    1013 +  
    1014 +      if args.long:
    1015 +          _render_long(entries, args, videos, video_rows, fields, colw, out)
    1016 +          return
    1017 +  
    1018 +      if args.one:
    1019 +          for e in entries:
    1020 +              line = e.name
    1021 +              row = video_rows.get(id(e))
    1022 +              if row:
    1023 +                  line = _join_video(e.name, namew, row, fields, colw)
    1024 +              print(line, file=out)
    1025 +          return
    1026 +  
    1027 +      # 默认版式：连续的非视频项攒成 run 走网格；视频每行独占
    1028 +      run: List[str] = []
    1029 +      width = terminal_width()
    1030 +      header_printed = False
    1031 +  
    1032 +      def flush_run() -> None:
    1033 +          nonlocal run
    1034 +          if run:
    1035 +              for line in format_grid(run, width):
    1036 +                  print(line, file=out)
    1037 +              run = []
    1038 +  
    1039 +      for e in entries:
    1040 +          row = video_rows.get(id(e))
    1041 +          if row is None:
    1042 +              run.append(e.name)
    1043 +              continue
    1044 +          flush_run()
    1045 +          if len(videos) >= 2 and not header_printed:
    1046 +              print(_header_line(namew, fields, colw), file=out)
    1047 +              header_printed = True
    1048 +          print(_join_video(e.name, namew, row, fields, colw), file=out)
    1049 +      flush_run()

    1052 +  def _header_line(namew: int, fields: Sequence[str], colw: Dict[str, int]) -> str:
    1053 +      parts = [pad_display("", namew)]
    1054 +      for key in fields:
    1055 +          parts.append(pad_display(FIELD_HEADER[key], colw[key], FIELD_ALIGN[key]))
    1056 +      return "  ".join(parts).rstrip()

    1059 +  def _join_video(name: str, namew: int, row: Dict[str, str],
    1060 +                  fields: Sequence[str], colw: Dict[str, int]) -> str:
    1061 +      if "res" in row and len(row) == 1:        # 探测失败
    1062 +          return f"{pad_display(name, namew)}  {row['res']}"
    1063 +      parts = [pad_display(name, namew)]
    1064 +      for key in fields:
    1065 +          parts.append(pad_display(row.get(key, ""), colw[key], FIELD_ALIGN[key]))
    1066 +      return "  ".join(parts).rstrip()

    1069 +  def _render_long(entries: List[Entry], args: "Options", videos: List[Entry],
    1070 +                   video_rows: Dict[int, Dict[str, str]], fields: Sequence[str],
    1071 +                   colw: Dict[str, int], out) -> None:
    1072 +      now = time.time()
    1073 +      nlinkw = max([len(str(e.st.st_nlink)) for e in entries], default=1)
    1074 +      sizes: List[str] = []
    1075 +      for e in entries:
    1076 +          if statmod.S_ISDIR(e.st.st_mode):
    1077 +              sizes.append("-")
    1078 +          elif args.human:
    1079 +              sizes.append(human_size(e.st.st_size))
    1080 +          else:
    1081 +              sizes.append(str(e.st.st_size))
    1082 +      sizew = max([len(s) for s in sizes], default=1)
    1083 +  
    1084 +      owners = [_user_name(e.st.st_uid) for e in entries]
    1085 +      groups = [_group_name(e.st.st_gid) for e in entries]
    1086 +      ownerw = max([disp_width(x) for x in owners], default=1)
    1087 +      groupw = max([disp_width(x) for x in groups], default=1)
    1088 +      namew = max([disp_width(e.name) for e in entries], default=0)
    1089 +  
    1090 +      header_printed = False
    1091 +      for i, e in enumerate(entries):
    1092 +          mode = _filemode(e.st)
    1093 +          head = (f"{mode} {e.st.st_nlink:>{nlinkw}} "
    1094 +                  f"{pad_display(owners[i], ownerw)} {pad_display(groups[i], groupw)} "
    1095 +                  f"{sizes[i]:>{sizew}} {_fmt_mtime(e.st.st_mtime, now)} "
    1096 +                  f"{pad_display(e.name, namew)}")
    1097 +          row = video_rows.get(id(e))
    1098 +          if row is None:
    1099 +              print(head.rstrip(), file=out)
    1100 +              continue
    1101 +          if len(videos) >= 2 and not header_printed:
    1102 +              print(pad_display("", len(head)) + "  " +
    1103 +                    "  ".join(pad_display(FIELD_HEADER[k], colw[k], FIELD_ALIGN[k])
    1104 +                              for k in fields), file=out)
    1105 +              header_printed = True
    1106 +          cells = "  ".join(pad_display(row.get(k, ""), colw[k], FIELD_ALIGN[k])
    1107 +                            for k in fields)
    1108 +          print(f"{head.rstrip()}  {cells}".rstrip(), file=out)

    10 +  ## 一、coreutils `ls` 的列布局（想逐字节一致就必须照抄）
    11 +  
    12 +  - **竖着填**（column-major）：`rows = ceil(N / cols)`，第 `c` 列放 `[c*rows, (c+1)*rows)`。
    13 +  - 列宽 = 该列**显示宽**最大值，下限 `MIN_COLUMN_WIDTH = 3`；**列间固定 2 列**（最后一列不加）。
    14 +  - ���「最大可能列数」往下试，第一个满足 `总宽 < 终端宽`（**严格小于**）的布局胜出；
    15 +    连 2 列都放不下就退化成每行一个。
    16 +  - **列间填充用 Tab 跳制表位**（8 的倍数），余量再补空格 —— `ls | cat -A` 能看到 `^I`。
    17 +    实测：`dir_a`(5) 补到 54 是 6 个 Tab + 6 个空格，不是 49 个空格。
    18 +    **Why:** 不走这套逻辑，输出看着一样但字节不同，diff 全红。
    19 +  - 非 TTY 时 `ls` 会退化成每行一个；`vidls` **故意不退化**（仍按 80 列铺网格），
    20 +    这样管道里也是紧凑的。已确认这是有意偏离。
    21 +  - `os.get_terminal_size()` 在 pty 没设尺寸时会返回 **columns=0**，必须当成失败回退到
    22 +    `$COLUMNS` / 80，否则整个布局退化成每行一个。

    24 +  ## 二、`-l` 长格式的细节
    25 +  
    26 +  - 表头是 `total N`（`N = sum(st_blocks) // 2`）。跟随 locale 输出 `total`（本机 `LANG=en_US.UTF-8`）。
    27 +  - 月份缩写要用 `strftime("%b")` 并 `setlocale(LC_TIME, "")`，与 `ls` 共用同一套 locale 数据；
    28 +    硬编码 `Jan/Feb/...` 在非 C locale 下会与 `ls` 不一致。
    29 +  - `-h` 人类可读：base 1024、**向上取整**（coreutils 是 ceiling），`<10` 保留 1 位小数
    30 +    （`3000 → 3.0K`，`88356 → 87K`）。
    31 +  - 时间：半年内（含未来时间）→ `%b %e %H:%M`，否则 → `%b %e  %Y`（两个空格）。

    35 +  ## 三、帧数四档来源（差异很大，必须带标签）
